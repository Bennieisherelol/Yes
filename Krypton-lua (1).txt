local dragging, dragInput, dragStart, startPos
local isDraggingAction = false
local draggableEnabled = true

robloxTopBarButton.InputBegan:Connect(function(input)
	if not draggableEnabled then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		isDraggingAction = false
		dragStart = input.Position
		startPos = robloxTopBarButton.Position
		
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

robloxTopBarButton.InputChanged:Connect(function(input)
	if not draggableEnabled then return end
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		dragInput = input
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not draggableEnabled then return end
	if input == dragInput and dragging then
		local delta = input.Position - dragStart
		if delta.Magnitude > 5 then
			isDraggingAction = true
		end
		robloxTopBarButton.Position = UDim2.new(
			startPos.X.Scale, 
			startPos.X.Offset + delta.X, 
			startPos.Y.Scale, 
			startPos.Y.Offset + delta.Y
		)
	end
end)

local function updateLineNumbers()
	local text = editorBox.Text
	local count = 1
	for _ in text:gmatch("\n") do
		count = count + 1
	end
	if text:sub(-1) == "\n" then
		count = count + 1
	end
	
	local totalHeight = math.max(5000, count * 18)
	editorBox.Size = UDim2.new(0.92, 0, 0, totalHeight)
	lineNumLabel.Size = UDim2.new(0.05, 0, 0, totalHeight)
	
	local lines = {}
	for i = 1, count do
		table.insert(lines, tostring(i) .. ". ")
	end
	lineNumLabel.Text = table.concat(lines, "\n")
end

local function saveThemeData(themeName)
	pcall(function()
		if writefile then
			writefile(THEME_SAVE_FILE, HttpService:JSONEncode({theme = themeName}))
		end
	end)
end

local function applyTheme(colorSeq, saveName)
	currentThemeSeq = colorSeq
	if saveName then
		saveThemeData(saveName)
	end
	pcall(function()
		for _, descendant in ipairs(screenGui:GetDescendants()) do
			if descendant:IsA("UIGradient") and descendant.Name ~= "TabGradient" and descendant.Parent ~= robloxTopBarButton and descendant.Parent ~= G2L["37"] then
				descendant.Color = colorSeq
			end
		end
	end)
end

pcall(function()
	if readfile and isfile and isfile(THEME_SAVE_FILE) then
		local data = HttpService:JSONDecode(readfile(THEME_SAVE_FILE))
		if data and data.theme and themePresets[data.theme] then
			applyTheme(themePresets[data.theme], nil)
		end
	end
end)

screenGui.DescendantAdded:Connect(function(descendant)
	if descendant:IsA("UIGradient") and descendant.Name ~= "TabGradient" and descendant.Parent ~= robloxTopBarButton and descendant.Parent ~= G2L["37"] then
		descendant.Color = currentThemeSeq
	end
end)

local SAVE_FILE_NAME = "KryptonAndroid_TabsData.json"
local tabs = {}

local function saveTabsData()
	pcall(function()
		if writefile then
			local saveData = {}
			for _, t in ipairs(tabs) do
				table.insert(saveData, {
					id = t.id,
					name = t.name,
					code = t.code
				})
			end
			writefile(SAVE_FILE_NAME, HttpService:JSONEncode(saveData))
		end
	end)
end

local function loadTabsData()
	local success, result = pcall(function()
		if readfile and isfile and isfile(SAVE_FILE_NAME) then
			return HttpService:JSONDecode(readfile(SAVE_FILE_NAME))
		end
	end)
	if success and type(result) == "table" then
		return result
	end
	return nil
end

local function switchTab(targetTab)
	for _, t in ipairs(tabs) do
		if t.button == targetTab then
			activeTab = t.button
			editorBox.Text = t.code or ""
			t.button.BackgroundTransparency = 0
			t.text.TextColor3 = Color3.fromRGB(255, 255, 255)
			if t.close then
				t.close.Visible = true
			end
		else
			t.button.BackgroundTransparency = 1
			t.text.TextColor3 = Color3.fromRGB(71, 88, 135)
			if t.close then
				t.close.Visible = false
			end
		end
	end
end

	for _, t in ipairs(tabs) do

			t.code = editorBox.Text
			saveTabsData()

	end
end)

local function getLowestAvailableId()
	local usedIds = {}
	for _, t in ipairs(tabs) do
		usedIds[t.id] = true
	end
	local id = 2
	while usedIds[id] do
		id = id + 1
	end
	return id
end

local function setupTabConnections(tabObj, isFirst)
	tabObj.button.MouseButton1Click:Connect(function()
		switchTab(tabObj.button)
	end)
	
	if not isFirst and tabObj.close then
		tabObj.close.MouseButton1Click:Connect(function()
			if #tabs > 1 then
				local indexToRemove = nil
				for i, t in ipairs(tabs) do
					if t.button == tabObj.button then
						indexToRemove = i
						break
					end
				end
				if indexToRemove then
					table.remove(tabs, indexToRemove)
					tabObj.button:Destroy()
					
					for _, t in ipairs(tabs) do
						if t.close then
							t.close.Visible = (t.button == activeTab)
						end
					end
					
					if activeTab == tabObj.button and #tabs > 0 then
						switchTab(tabs[1].button)
					end
					saveTabsData()
				end
			end
		end)
	end
end

local function createTabUI(id, tabName, codeContent, isFirst)
	local clonedTab
	if isFirst then
		clonedTab = originalTab
		local firstCloseBtn = clonedTab:FindFirstChild("CloseTab")
		if firstCloseBtn then
			firstCloseBtn:Destroy()
		end
	else
		clonedTab = originalTab:Clone()
		clonedTab.Name = "Tab_" .. id
		clonedTab.BackgroundTransparency = 1
		
		local grad = clonedTab:FindFirstChildOfClass("UIGradient")
		if not grad then
			grad = G2L["27"]:Clone()
			grad.Parent = clonedTab
		end
		
		clonedTab.Parent = tabsScrollingFrame
	end
	
	clonedTab.LayoutOrder = id
	
	local textLbl = clonedTab:WaitForChild("TextLabel")
	local closeBtn = clonedTab:FindFirstChild("CloseTab")
	
	if not isFirst then
		if not closeBtn then
			closeBtn = Instance.new("ImageButton", clonedTab)
			closeBtn.Name = "CloseTab"
			closeBtn.BackgroundTransparency = 1
			closeBtn.BorderSizePixel = 0
			closeBtn.Size = UDim2.new(0.16821, 0, 0.4, 0)
			closeBtn.Position = UDim2.new(0.73694, 0, 0.3, 0)
			closeBtn.Image = "rbxassetid://80433169334469"
			closeBtn.ImageColor3 = Color3.fromRGB(0, 0, 0)
		end
		closeBtn.Visible = false
	else
		closeBtn = nil
	end
	
	textLbl.Text = tabName
	textLbl.TextColor3 = isFirst and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(71, 88, 135)
	
	local tabObj = {
		id = id,
		name = tabName,
		button = clonedTab,
		text = textLbl,
		close = closeBtn,
		code = codeContent or ""
	}
	
	table.insert(tabs, tabObj)
	setupTabConnections(tabObj, isFirst)
	return tabObj
end

local LOCAL_SCRIPTS_FILE = "KryptonAndroid_LocalCategories.json"
localScriptsCache = { Saved = {}, Autoexec = {}, Favorites = {} }

loadLocalScriptsData = function()
	pcall(function()
		if readfile and isfile and isfile(LOCAL_SCRIPTS_FILE) then
			local decoded = HttpService:JSONDecode(readfile(LOCAL_SCRIPTS_FILE))
			if type(decoded) == "table" then
				localScriptsCache = decoded
			end
		end
	end)
	if not localScriptsCache.Saved then localScriptsCache.Saved = {} end
	if not localScriptsCache.Autoexec then localScriptsCache.Autoexec = {} end
	if not localScriptsCache.Favorites then localScriptsCache.Favorites = {} end
	
	local modified = false
	for _, catName in ipairs({"Saved", "Autoexec", "Favorites"}) do
		if localScriptsCache[catName] then
			for _, item in ipairs(localScriptsCache[catName]) do
				if not item.id then
					item.id = tostring(tick()) .. "_" .. math.random(1000, 9999)
					modified = true
				end
			end
		end
	end
	if modified and writefile then
		pcall(function()
			writefile(LOCAL_SCRIPTS_FILE, HttpService:JSONEncode(localScriptsCache))
		end)
	end
	
	return localScriptsCache
end

saveLocalScriptsData = function()
	pcall(function()
		if writefile then
			writefile(LOCAL_SCRIPTS_FILE, HttpService:JSONEncode(localScriptsCache))
		end
	end)
end

local function fetchWebRequest(url)
	local response
	pcall(function()
		if syn and syn.request then
			local res = syn.request({ Url = url, Method = "GET" })
			if res and res.Body then response = res.Body end
		elseif http_request then
			local res = http_request({ Url = url, Method = "GET" })
			if res and res.Body then response = res.Body end
		elseif request then
			local res = request({ Url = url, Method = "GET" })
			if res and res.Body then response = res.Body end
		else
			response = game:HttpGet(url)
		end
	end)
	return response
end

fetchCloudScriptsForProvider = function(provider, searchQuery)
	clearCloudResults()
	searchQuery = searchQuery or ""
	
	if provider == "Local" then
		task.spawn(function()
			local data = loadLocalScriptsData()
			local list = data[activeLocalCategory] or {}
			if #list > 0 then
				for _, item in ipairs(list) do
					if not item.id then
						item.id = tostring(tick()) .. "_" .. math.random(1000, 9999)
					end
					if searchQuery == "" or item.name:lower():find(searchQuery:lower(), 1, true) then
						addLocalScriptCard(item.name, item.code, activeLocalCategory, item.id)
					end
				end
			else
				local emptyLbl = Instance.new("TextLabel", cloudScrollingFrame)
				emptyLbl.BackgroundTransparency = 1
				emptyLbl.Size = UDim2.new(1, 0, 0, 100)
				emptyLbl.Font = Enum.Font.SourceSansBold
				emptyLbl.Text = "Oops! No scripts found"
				emptyLbl.TextColor3 = Color3.fromRGB(255, 255, 255)
				emptyLbl.TextSize = 16
				emptyLbl.TextXAlignment = Enum.TextXAlignment.Center
				emptyLbl.TextYAlignment = Enum.TextYAlignment.Center
			end
		end)
		return
	end
	
	task.spawn(function()
		if provider == "ScriptBlox" then
			pcall(function()
				local url = "https://scriptblox.com/api/script/fetch"
				if searchQuery ~= "" then
					url = "https://scriptblox.com/api/script/search?q=" .. HttpService:UrlEncode(searchQuery)
				end
				local response = fetchWebRequest(url)
				if response then
					local successDecode, data = pcall(function() return HttpService:JSONDecode(response) end)
					if successDecode and data then
						local scriptList = data.result and data.result.scripts or data.scripts
						if type(scriptList) == "table" then
							for _, script in ipairs(scriptList) do
								local title = script.title or "Script"
								local rawContent = script.script or ('loadstring(game:HttpGet("https://scriptblox.com/raw/' .. tostring(script.slug or "") .. '"))()')
								addCloudScriptCard(title, rawContent, "ScriptBlox", script)
							end
						end
					end
				end
			end)
		elseif provider == "Rscripts" then
			pcall(function()
				local rUrls = {
					"https://rscripts.net/api/v2/scripts",
					"https://rscripts.net/api/v1/scripts",
					"https://rscripts.net/api/scripts"
				}
				if searchQuery ~= "" then
					rUrls = {
						"https://rscripts.net/api/v2/scripts?q=" .. HttpService:UrlEncode(searchQuery),
						"https://rscripts.net/api/v1/scripts?q=" .. HttpService:UrlEncode(searchQuery),
						"https://rscripts.net/api/scripts?search=" .. HttpService:UrlEncode(searchQuery)
					}
				end
				
				local loaded = false
				for _, rUrl in ipairs(rUrls) do
					local rResponse = fetchWebRequest(rUrl)
					if rResponse then
						local successDecode, rData = pcall(function() return HttpService:JSONDecode(rResponse) end)
						if successDecode and rData then
							local rList = rData.scripts or rData.data or rData
							if type(rList) == "table" and #rList > 0 then
								for _, script in ipairs(rList) do
									local title = script.title or script.name or "Rscript"
									local rawContent = script.script or script.code or ('loadstring(game:HttpGet("https://rscripts.net/raw/' .. tostring(script.id or script.slug or "") .. '"))()')
									addCloudScriptCard(title, rawContent, "Rscripts", script)
								end
								loaded = true
								break
							end
						end
					end
				end
				
				if not loaded then
					addCloudScriptCard("No Rscripts Found", "-- Rscripts API endpoint was unreachable or returned empty data", "Rscripts", nil)
				end
			end)
		elseif provider == "HaxHell" then
			pcall(function()
				local hUrls = {
					"https://haxhell.com/api/scripts",
					"https://haxhell.com/api/v1/scripts"
				}
				if searchQuery ~= "" then
					hUrls = {
						"https://haxhell.com/api/scripts?search=" .. HttpService:UrlEncode(searchQuery),
						"https://haxhell.com/api/v1/scripts?q=" .. HttpService:UrlEncode(searchQuery)
					}
				end
				
				for _, hUrl in ipairs(hUrls) do
					local hResponse = fetchWebRequest(hUrl)
					if hResponse then
						local successDecode, hData = pcall(function() return HttpService:JSONDecode(hResponse) end)
						if successDecode and hData then
							local hList = hData.scripts or hData.data or hData
							if type(hList) == "table" then
								for _, script in ipairs(hList) do
									local title = script.title or script.name or "HaxHell Script"
									local rawContent = script.script or script.code or script.content or ""
									addCloudScriptCard(title, rawContent, "HaxHell", script)
								end
								break
							end
						end
					end
				end
			end)
		end
	end)
end

			id = tostring(tick()) .. "_" .. math.random(1000, 9999),
			name = sName,
			code = sCode
		}
		table.insert(data[activeLocalCategory], newScript)
		saveLocalScriptsData()
		modalOverlay.Visible = false
		fetchCloudScriptsForProvider("Local", "")
	end
end)
